Create `components/components.js` — the fetch-based component loader. This script **must NOT** use `defer`, `async`, or `type="module"` on its `<script>` tag (document this in a header comment).

**Implementation — exact structure:**

```javascript
/**
 * components.js — Shared component loader for Meu Planetinha
 *
 * Fetches nav.html and footer.html fragments, injects them into
 * [data-component] placeholder elements, rewrites links, detects
 * active page, and sets up the mobile menu.
 *
 * IMPORTANT: The <script> tag loading this file must NOT have
 * defer, async, or type="module" attributes.
 * document.currentScript is captured synchronously at parse time.
 *
 * Usage in HTML:
 *   <div data-component="nav"></div>
 *   <div data-component="footer"></div>
 *   <script src="components/components.js"></script>
 */

// === 1. BASE PATH DETECTION ===
// Capture script URL at parse time (before DOMContentLoaded)
const SCRIPT_URL = document.currentScript
  ? document.currentScript.src
  : '';
// Strip 'components/components.js' to get the site base URL
const BASE_URL = SCRIPT_URL.substring(
  0,
  SCRIPT_URL.lastIndexOf('components/')
);
```

**Section 2 — Component fetcher:**

```javascript
// === 2. COMPONENT FETCHER ===
async function loadComponent(name) {
  const url = BASE_URL + 'components/' + name + '.html';
  try {
    const response = await fetch(url);
    if (!response.ok) return null;
    return await response.text();
  } catch (e) {
    // Silent failure per design decision — no error banner
    return null;
  }
}
```

**Section 3 — Link rewriting:**

```javascript
// === 3. LINK REWRITING ===
function rewriteLinks(container) {
  const links = container.querySelectorAll('[data-href]');
  links.forEach(function(link) {
    const dataHref = link.getAttribute('data-href');
    // Skip external and special protocol links
    if (/^(https?:|mailto:|tel:)/.test(dataHref)) {
      link.setAttribute('href', dataHref);
    } else {
      link.setAttribute('href', BASE_URL + dataHref);
    }
  });
}
```

**Section 4 — Active page detection:**

```javascript
// === 4. ACTIVE PAGE DETECTION ===
function markActivePage(navElement) {
  const currentPath = normalizePath(location.pathname);
  const links = navElement.querySelectorAll('.nav-links a, .nav-overlay-links a');

  links.forEach(function(link) {
    const linkPath = normalizePath(new URL(link.href, location.origin).pathname);
    if (linkPath === currentPath) {
      link.setAttribute('aria-current', 'page');
    }
  });
}

function normalizePath(pathname) {
  // Strip trailing slash
  let p = pathname.replace(/\/+$/, '');
  // Treat /index.html as the directory
  p = p.replace(/\/index\.html$/, '');
  // If empty after stripping, it's the root
  if (!p) p = '/';
  return p.toLowerCase();
}
```

**Section 5 — Mobile menu toggle:**

```javascript
// === 5. MOBILE MENU ===
function setupMobileMenu(navElement) {
  var hamburger = navElement.querySelector('.nav-hamburger');
  var overlay = navElement.querySelector('.nav-overlay');
  var closeBtn = navElement.querySelector('.nav-overlay-close');

  if (!hamburger || !overlay) return;

  function openMenu() {
    overlay.classList.add('is-open');
    overlay.setAttribute('aria-hidden', 'false');
    hamburger.setAttribute('aria-expanded', 'true');
    document.body.style.overflow = 'hidden';
    // Focus the close button for accessibility
    if (closeBtn) closeBtn.focus();
  }

  function closeMenu() {
    overlay.classList.remove('is-open');
    overlay.setAttribute('aria-hidden', 'true');
    hamburger.setAttribute('aria-expanded', 'false');
    document.body.style.overflow = '';
    // Return focus to hamburger
    hamburger.focus();
  }

  hamburger.addEventListener('click', function() {
    var isOpen = overlay.classList.contains('is-open');
    if (isOpen) {
      closeMenu();
    } else {
      openMenu();
    }
  });

  if (closeBtn) {
    closeBtn.addEventListener('click', closeMenu);
  }

  // Close on Escape key
  overlay.addEventListener('keydown', function(e) {
    if (e.key === 'Escape') {
      closeMenu();
    }
  });

  // Close when clicking overlay background (not the links/buttons)
  overlay.addEventListener('click', function(e) {
    if (e.target === overlay) {
      closeMenu();
    }
  });
}
```

**Section 6 — Initialization (runs on DOMContentLoaded):**

```javascript
// === 6. INIT ===
document.addEventListener('DOMContentLoaded', async function() {
  var placeholders = document.querySelectorAll('[data-component]');

  for (var i = 0; i < placeholders.length; i++) {
    var el = placeholders[i];
    var name = el.getAttribute('data-component');
    var html = await loadComponent(name);
    if (html) {
      el.innerHTML = html;
      rewriteLinks(el);

      // Post-injection setup
      if (name === 'nav') {
        markActivePage(el);
        setupMobileMenu(el);
      }
    } else if (name === 'nav') {
      // Fallback: inject minimal inline nav when fetch fails
      el.innerHTML =
        '<nav class="site-nav" aria-label="Navegação principal">' +
        '<div class="nav-container">' +
        '<a class="nav-logo" href="' + BASE_URL + 'index.html">🪐 Meu Planetinha</a>' +
        '<div class="nav-links">' +
        '<a href="' + BASE_URL + 'explorar/explorar.html">Explorar Jogos</a>' +
        '<a href="' + BASE_URL + 'sobre_nos/sobre_nos.html">Sobre Nós</a>' +
        '</div></div></nav>';
    }
  }
});
```

**Critical implementation notes:**

1. **`document.currentScript` fallback:** If `document.currentScript` is `null` (shouldn't happen with correct `<script>` tag), log a console warning: `console.warn('[components.js] document.currentScript is null — ensure <script> tag has no defer/async/type="module"')`.

2. **Sequential fetching:** The `for` loop with `await` processes components sequentially. This is intentional — nav should be injected before footer to maintain DOM order. If you want parallel fetching for speed, use `Promise.all` to fetch both HTML fragments simultaneously, then inject in order. Either approach is acceptable.

3. **No ES6 modules, no arrow functions in critical paths:** Use `function()` declarations for maximum compatibility. Arrow functions are fine for internal utility but the code should work in all modern evergreen browsers.

4. **`<noscript>` fallback in page templates:** Each page template includes a `<noscript>` block inside `[data-component="nav"]` with a minimal hardcoded nav. This covers the JS-disabled case. When JS runs but fetch fails, the `loadComponent()` fallback branch injects a similar minimal nav.

5. **The `rewriteLinks` function must handle the `mailto:` link in the footer** — it checks for protocol prefixes and sets `href` directly without prepending `BASE_URL`.
